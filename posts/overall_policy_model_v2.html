1 Overview

This model builds on Overall Policy Model v1 with five layered scenarios evaluated as a spend-vs-benefit frontier. Each scenario adds one policy lever; the primary outputs are (i) net monetary benefit (NMB) — the societal cost saving net of regulator spend — and (ii) regulator cost per death averted. Both panels appear side-by-side in every line graph.

Scenario Label New lever added
1.1 Administrative surveillance (S1 + S2) PIH MVD feed and BChE auto-notify
1.2 + Coordination (C1–C5) NMC closure-of-loop, DALRRD referral, intel feed, clinician feedback, KPI dashboard
1.3 + Clustering Each EHP investigation surfaces hidden cluster members
1.4 + Community (S1b) CHW/EHP event-based community layer; FP sensitivity
1.5 + Chemical agent phase-out Terbufos (T2) and aldicarb; agent attribution from S3 sentinel toxicology

Key structural results maintained from v1:

  1. Detection–workload coupling — false-positive-inflated alert volume drives induced EHP and DALRRD workload; marginal returns diminish once community surveillance is added.
  2. Source-control coupling — chemical restrictions reduce true cases and deaths, lowering both the health/YLL bill and induced workload by the same fraction.

# ── Shared model function ─────────────────────────────────────────────────────
#
# Parameters:
#   d             per-case detection probability
#   c_surv        annual surveillance operating cost (R/yr)
#   c_coord       annual coordination operating cost (R/yr)
#   p_fp          false-positive alert rate (share of gross alerts that are not true cases)
#   k_cluster     cluster multiplier [1 = no clustering]
#   coord_dynamic if TRUE, scale coordination deaths-averted with (d - d_SQ)
#   terb_phase    fraction of terbufos-attributable burden retired [0,1]
#   f_other       attributable fraction of a second chemical agent (e.g. aldicarb)
#   phi_other     fraction of other-agent burden retired under its restriction
#   c_regulatory  DALRRD/Dept. Agriculture enforcement spend (R/yr)
#
run_scenario_v2 <- function(
    d,
    c_surv,
    c_coord       = 0,
    p_fp          = 0,
    k_cluster     = 1.0,
    coord_dynamic = FALSE,
    terb_phase    = 0,
    f_other       = 0,
    phi_other     = 0,
    c_regulatory  = 0
) {

  # (A) True burden after source-control retirements
  f_retired    <- f_terb_mid * terb_phase + f_other * phi_other
  N_true_eff   <- N_true   * (1 - f_retired)
  N_deaths_eff <- N_deaths * (1 - f_retired)
  cfr_eff      <- N_deaths_eff / N_true_eff

  # (B) Detection and alerts
  detected_true <- d * N_true_eff
  gross_alerts  <- if_else(p_fp < 1, detected_true / (1 - p_fp), detected_true * 2)
  false_alerts  <- gross_alerts - detected_true

  # (C) Clustering: each confirmed investigation surfaces k_cluster - 1 additional invisible cases
  cluster_bonus <- detected_true * (k_cluster - 1)   # extra cases surfaced by field investigation

  # (D) Induced workload — scales with GROSS alerts (system cannot distinguish FP until investigated)
  c_investigation <- gross_alerts * c_per_investigation
  c_dalrrd_load   <- gross_alerts * p_dalrrd_action * c_per_dalrrd_action

  # (E) Deaths averted decomposition
  #   (i) Surveillance + response: incremental detection above SQ drives source removal / clinical response
  deaths_av_surv    <- eta * (d - d_SQ) * N_true_eff * cfr_eff

  #   (ii) Clustering bonus: invisible cluster members surfaced and treated
  deaths_av_cluster <- cluster_bonus * cfr_eff * eta

  #   (iii) Coordination: fixed CSV param OR dynamic (scaled by incremental detection vs MVP baseline)
  if (coord_dynamic && c_coord > 0) {
    # Scale coordination deaths-averted linearly with detection increment above SQ
    # Anchored at: d_MVP → n_av_coord_FP deaths averted
    deaths_av_coord <- n_av_coord_FP * max(0, (d - d_SQ)) / (d_MVP - d_SQ)
  } else if (c_coord > 0) {
    deaths_av_coord <- n_av_coord_FP * (c_coord / c_coord_FP)
  } else {
    deaths_av_coord <- 0
  }

  #   (iv) Source control (terbufos + other agent)
  deaths_av_terb  <- N_deaths * f_terb_mid * terb_phase * phi_terb
  deaths_av_other <- N_deaths * f_other    * phi_other

  deaths_av_total <- deaths_av_surv + deaths_av_cluster + deaths_av_coord +
                     deaths_av_terb + deaths_av_other

  # (F) Health and YLL bill under this scenario
  c_health <- N_true_eff   * c_morb
  c_yll    <- N_deaths_eff * c_mort

  # (G) Regulator and societal totals
  c_regulator <- c_surv + c_coord + c_investigation + c_dalrrd_load + c_regulatory
  c_society   <- c_regulator + c_health + c_yll

  # (H) Net monetary benefit (NMB) = societal cost saving vs status quo, net of regulator spend
  # NMB > 0: policy saves more than it costs from a whole-of-society perspective
  NMB <- c_society_SQ - c_society

  tibble(
    detected_true     = detected_true,
    cluster_bonus     = cluster_bonus,
    gross_alerts      = gross_alerts,
    false_alerts      = false_alerts,
    deaths_av_surv    = deaths_av_surv,
    deaths_av_cluster = deaths_av_cluster,
    deaths_av_coord   = deaths_av_coord,
    deaths_av_terb    = deaths_av_terb,
    deaths_av_other   = deaths_av_other,
    deaths_averted    = deaths_av_total,
    c_surv            = c_surv,
    c_coord           = c_coord,
    c_investigation   = c_investigation,
    c_dalrrd_load     = c_dalrrd_load,
    c_regulatory      = c_regulatory,
    c_regulator_tot   = c_regulator,
    c_health          = c_health,
    c_yll             = c_yll,
    c_society_tot     = c_society,
    NMB               = NMB,
    cost_per_death_av = if_else(deaths_av_total > 0,
                                c_regulator / deaths_av_total, NA_real_)
  )
}

# ── Shared dual-panel plot helper ─────────────────────────────────────────────
# x_col, colour_col: bare column name strings (or NULL for no colour grouping)
dual_plot <- function(df,
                      x_col,
                      nmb_col   = "NMB",
                      cpd_col   = "cost_per_death_av",
                      colour_col = NULL,
                      x_lab     = "Annual regulator spend (R/yr)") {

  mk_aes <- function(y_col) {
    if (is.null(colour_col)) {
      aes(x = .data[[x_col]], y = .data[[y_col]])
    } else {
      aes(x = .data[[x_col]], y = .data[[y_col]], colour = .data[[colour_col]])
    }
  }

  p1 <- ggplot(df, mk_aes(nmb_col)) +
    geom_line(linewidth = 1.1) + geom_point(size = 2) +
    geom_hline(yintercept = 0, linetype = "dashed", alpha = 0.4) +
    scale_x_continuous(labels = comma_format(prefix = "R")) +
    scale_y_continuous(labels = comma_format(prefix = "R")) +
    labs(x = x_lab, y = "Net monetary benefit (R/yr)", colour = NULL) +
    theme_minimal(base_size = 11) +
    theme(legend.position = "bottom")

  p2 <- ggplot(df |> filter(!is.na(.data[[cpd_col]])), mk_aes(cpd_col)) +
    geom_line(linewidth = 1.1) + geom_point(size = 2) +
    scale_x_continuous(labels = comma_format(prefix = "R")) +
    scale_y_continuous(labels = comma_format(prefix = "R")) +
    labs(x = x_lab, y = "Regulator cost per death averted (R)", colour = NULL) +
    theme_minimal(base_size = 11) +
    theme(legend.position = "bottom")

  p1 + p2 + plot_layout(guides = "collect") & theme(legend.position = "bottom")
}

3 Scenario 1.1 — Administrative surveillance (S1 + S2)

S1 (PIH minimum viable dataset) and S2 (BChE auto-notify) each add one feed to NMC without introducing false-positive inflation — alerts come from confirmed laboratory or clinical events. This scenario shows the step-wise spend-vs-benefit for adding each feed and combines them into the Surveillance MVP.

Interpretation. Both S1 and S2 are individually cost-saving (NMB > 0) because the health/YLL burden saved exceeds the modest operating cost of each feed. S2 (BChE auto-notify) contributes more detection gain than S1 (PIH MVD) per rand because NHLS BChE tests a larger flow of severe cases. The MVP combination is the minimum investment that closes the administrative detection gap.

4 Scenario 1.2 — + Coordination (C1–C5)

Coordination adds five components (C1–C5) that convert surveillance notifications into downstream source-removal and enforcement actions. The table shows both the static CSV parameter (957 deaths averted at national scale at full-coordination package spend) and the dynamic formulation (deaths averted scaled linearly with detection increment above status quo).

Interpretation. Each coordination component is independently cost-saving. C3 (NICD→DALRRD quarterly intelligence feed and cluster-alert SOP) contributes the largest share of deaths averted per rand of coordination spend, because it converts confirmed alerts into regulatory enforcement action. The Full Package (MVP + C1–C5) costs ~R282k/yr total, averts ~957 deaths/yr, and yields a cost per death averted well below the human-capital threshold.

5 Scenario 1.3 — Clustering effect on community poisonings

EHP field investigations routinely encounter household or community events where a confirmed case is the first alert to a cluster of unreported exposures. The cluster multiplier k represents the average number of cases addressed per investigation. k = 1 means no clustering (current model assumption); k > 1 means each investigation surfaces additional invisible cases who get treated and whose deaths are partially averted.

This scenario evaluates the sensitivity of NMB and deaths averted to k across three values (k_lo = 1.2, k_mid = 2, k_hi = 3.5) at both the MVP and S1b detection levels.

Interpretation. Even modest clustering (k = 1.2) improves both the NMB and cost-per-death metrics. At the working value (k = 2) and MVP detection, clustering adds ~149 deaths averted relative to the no-clustering assumption. k should be measured directly in EHP field-investigation logs; the dotted vertical lines mark the lo/mid/hi values used in sensitivity tables.

6 Scenario 1.4 — Community surveillance (S1b) with false-positive sensitivity

The community/EHP tier (S1b) lifts detection beyond the administrative ceiling but introduces a false-positive rate that inflates gross alert volume and drives induced EHP and DALRRD workload. This scenario holds detection at d_S1b_mid = 40% and varies (i) S1b operating spend and (ii) the false-positive rate.

Interpretation. At the working FP rate (35%, dashed line), S1b mid-spend remains cost-saving (NMB > 0), but the margin is substantially narrower than the administrative MVP. Above ~50% FP the lower-spend S1b option becomes cost-losing. The FP rate is the dominant uncertainty in the S1b decision — a focused pre-rollout pilot measuring FP in 2–3 provinces would resolve this before national investment is committed.

7 Scenario 1.5 — Chemical agent phase-out (terbufos + aldicarb)

Source control is the only lever that reduces the denominator the system has to process. S3 sentinel toxicology provides the agent-attribution evidence base needed to quantify which banning actions are worth the enforcement spend. This scenario models terbufos (T2 restriction) and aldicarb (already informally banned; attribution uncertain) with varying attributable fractions.

Interpretation. At the working attributable fraction (terbufos 25% + aldicarb 12%), the combined phase-out averts ~573 deaths in year 1 and generates an NMB that grows as the ramp completes. S3 sentinel toxicology (R1.2M pilot, R3.6M at 5 sites) is the prerequisite for moving aldicarb attribution from f_aldi_mid = 12% (literature estimate) to a SA-measured value.

## Overview

This model builds on [Overall Policy Model v1](overall_policy_model.qmd) with five layered scenarios evaluated as a **spend-vs-benefit frontier**. Each scenario adds one policy lever; the primary outputs are *(i)* net monetary benefit (NMB) — the societal cost saving net of regulator spend — and *(ii)* regulator cost per death averted. Both panels appear side-by-side in every line graph.

| Scenario | Label | New lever added |
|---|---|---|
| 1.1 | Administrative surveillance (S1 + S2) | PIH MVD feed and BChE auto-notify |
| 1.2 | + Coordination (C1–C5) | NMC closure-of-loop, DALRRD referral, intel feed, clinician feedback, KPI dashboard |
| 1.3 | + Clustering | Each EHP investigation surfaces hidden cluster members |
| 1.4 | + Community (S1b) | CHW/EHP event-based community layer; FP sensitivity |
| 1.5 | + Chemical agent phase-out | Terbufos (T2) and aldicarb; agent attribution from S3 sentinel toxicology |

**Key structural results maintained from v1:**

1. *Detection–workload coupling* — false-positive-inflated alert volume drives induced EHP and DALRRD workload; marginal returns diminish once community surveillance is added.
2. *Source-control coupling* — chemical restrictions reduce true cases and deaths, lowering both the health/YLL bill and induced workload by the same fraction.

# ── Shared model function ─────────────────────────────────────────────────────
#
# Parameters:
#   d             per-case detection probability
#   c_surv        annual surveillance operating cost (R/yr)
#   c_coord       annual coordination operating cost (R/yr)
#   p_fp          false-positive alert rate (share of gross alerts that are not true cases)
#   k_cluster     cluster multiplier [1 = no clustering]
#   coord_dynamic if TRUE, scale coordination deaths-averted with (d - d_SQ)
#   terb_phase    fraction of terbufos-attributable burden retired [0,1]
#   f_other       attributable fraction of a second chemical agent (e.g. aldicarb)
#   phi_other     fraction of other-agent burden retired under its restriction
#   c_regulatory  DALRRD/Dept. Agriculture enforcement spend (R/yr)
#
run_scenario_v2 <- function(
    d,
    c_surv,
    c_coord       = 0,
    p_fp          = 0,
    k_cluster     = 1.0,
    coord_dynamic = FALSE,
    terb_phase    = 0,
    f_other       = 0,
    phi_other     = 0,
    c_regulatory  = 0
) {

  # (A) True burden after source-control retirements
  f_retired    <- f_terb_mid * terb_phase + f_other * phi_other
  N_true_eff   <- N_true   * (1 - f_retired)
  N_deaths_eff <- N_deaths * (1 - f_retired)
  cfr_eff      <- N_deaths_eff / N_true_eff

  # (B) Detection and alerts
  detected_true <- d * N_true_eff
  gross_alerts  <- if_else(p_fp < 1, detected_true / (1 - p_fp), detected_true * 2)
  false_alerts  <- gross_alerts - detected_true

  # (C) Clustering: each confirmed investigation surfaces k_cluster - 1 additional invisible cases
  cluster_bonus <- detected_true * (k_cluster - 1)   # extra cases surfaced by field investigation

  # (D) Induced workload — scales with GROSS alerts (system cannot distinguish FP until investigated)
  c_investigation <- gross_alerts * c_per_investigation
  c_dalrrd_load   <- gross_alerts * p_dalrrd_action * c_per_dalrrd_action

  # (E) Deaths averted decomposition
  #   (i) Surveillance + response: incremental detection above SQ drives source removal / clinical response
  deaths_av_surv    <- eta * (d - d_SQ) * N_true_eff * cfr_eff

  #   (ii) Clustering bonus: invisible cluster members surfaced and treated
  deaths_av_cluster <- cluster_bonus * cfr_eff * eta

  #   (iii) Coordination: fixed CSV param OR dynamic (scaled by incremental detection vs MVP baseline)
  if (coord_dynamic && c_coord > 0) {
    # Scale coordination deaths-averted linearly with detection increment above SQ
    # Anchored at: d_MVP → n_av_coord_FP deaths averted
    deaths_av_coord <- n_av_coord_FP * max(0, (d - d_SQ)) / (d_MVP - d_SQ)
  } else if (c_coord > 0) {
    deaths_av_coord <- n_av_coord_FP * (c_coord / c_coord_FP)
  } else {
    deaths_av_coord <- 0
  }

  #   (iv) Source control (terbufos + other agent)
  deaths_av_terb  <- N_deaths * f_terb_mid * terb_phase * phi_terb
  deaths_av_other <- N_deaths * f_other    * phi_other

  deaths_av_total <- deaths_av_surv + deaths_av_cluster + deaths_av_coord +
                     deaths_av_terb + deaths_av_other

  # (F) Health and YLL bill under this scenario
  c_health <- N_true_eff   * c_morb
  c_yll    <- N_deaths_eff * c_mort

  # (G) Regulator and societal totals
  c_regulator <- c_surv + c_coord + c_investigation + c_dalrrd_load + c_regulatory
  c_society   <- c_regulator + c_health + c_yll

  # (H) Net monetary benefit (NMB) = societal cost saving vs status quo, net of regulator spend
  # NMB > 0: policy saves more than it costs from a whole-of-society perspective
  NMB <- c_society_SQ - c_society

  tibble(
    detected_true     = detected_true,
    cluster_bonus     = cluster_bonus,
    gross_alerts      = gross_alerts,
    false_alerts      = false_alerts,
    deaths_av_surv    = deaths_av_surv,
    deaths_av_cluster = deaths_av_cluster,
    deaths_av_coord   = deaths_av_coord,
    deaths_av_terb    = deaths_av_terb,
    deaths_av_other   = deaths_av_other,
    deaths_averted    = deaths_av_total,
    c_surv            = c_surv,
    c_coord           = c_coord,
    c_investigation   = c_investigation,
    c_dalrrd_load     = c_dalrrd_load,
    c_regulatory      = c_regulatory,
    c_regulator_tot   = c_regulator,
    c_health          = c_health,
    c_yll             = c_yll,
    c_society_tot     = c_society,
    NMB               = NMB,
    cost_per_death_av = if_else(deaths_av_total > 0,
                                c_regulator / deaths_av_total, NA_real_)
  )
}

# ── Shared dual-panel plot helper ─────────────────────────────────────────────
# x_col, colour_col: bare column name strings (or NULL for no colour grouping)
dual_plot <- function(df,
                      x_col,
                      nmb_col   = "NMB",
                      cpd_col   = "cost_per_death_av",
                      colour_col = NULL,
                      x_lab     = "Annual regulator spend (R/yr)") {

  mk_aes <- function(y_col) {
    if (is.null(colour_col)) {
      aes(x = .data[[x_col]], y = .data[[y_col]])
    } else {
      aes(x = .data[[x_col]], y = .data[[y_col]], colour = .data[[colour_col]])
    }
  }

  p1 <- ggplot(df, mk_aes(nmb_col)) +
    geom_line(linewidth = 1.1) + geom_point(size = 2) +
    geom_hline(yintercept = 0, linetype = "dashed", alpha = 0.4) +
    scale_x_continuous(labels = comma_format(prefix = "R")) +
    scale_y_continuous(labels = comma_format(prefix = "R")) +
    labs(x = x_lab, y = "Net monetary benefit (R/yr)", colour = NULL) +
    theme_minimal(base_size = 11) +
    theme(legend.position = "bottom")

  p2 <- ggplot(df |> filter(!is.na(.data[[cpd_col]])), mk_aes(cpd_col)) +
    geom_line(linewidth = 1.1) + geom_point(size = 2) +
    scale_x_continuous(labels = comma_format(prefix = "R")) +
    scale_y_continuous(labels = comma_format(prefix = "R")) +
    labs(x = x_lab, y = "Regulator cost per death averted (R)", colour = NULL) +
    theme_minimal(base_size = 11) +
    theme(legend.position = "bottom")

  p1 + p2 + plot_layout(guides = "collect") & theme(legend.position = "bottom")
}

## Scenario 1.3 — Clustering effect on community poisonings

## Scenario 1.4 — Community surveillance (S1b) with false-positive sensitivity

#| echo: false
#| message: false
#| warning: false

yll_per_death   <- pick("yll_per_death_gbd")          # 44 yr (provisional)
yld_severe      <- pick("yld_per_case_severe")        # 0.033 yr/case
yld_mild        <- pick("yld_per_case_mild")          # 0.003 yr/case
p_sev           <- pick("p_severe_among_detected")    # 0.45
icer            <- pick("icer_per_daly_gdp")          # R110,000/DALY
c_mort_daly     <- pick("c_mort_daly_icer")           # R4,840,000/death
c_morb_daly_sev <- pick("c_morb_daly_icer_severe")    # R3,630/severe case
c_morb_daly_mil <- pick("c_morb_daly_icer_mild")      # R330/mild case

# Blended per-case YLD cost (weighted severe/mild mix)
c_morb_daly_blended <- p_sev * c_morb_daly_sev + (1 - p_sev) * c_morb_daly_mil

# Status-quo DALY burden (societal reference under ICER approach)
daly_sq <- N_deaths * yll_per_death + N_true * (p_sev * yld_severe + (1 - p_sev) * yld_mild)
c_society_SQ_daly <- N_true * c_morb_daly_blended + N_deaths * c_mort_daly

# ── DALY-based scenario runner wrapper ────────────────────────────────────────
# Runs run_scenario_v2() with DALY-ICER costs instead of human-capital costs,
# then appends DALY summary columns.
run_scenario_daly <- function(...) {
  # Temporarily override c_morb / c_mort via re-implementation
  # (avoids touching the shared function signature)
  r <- run_scenario_v2(...)
  N_true_eff   <- r$c_health / c_morb    # back-derive effective burden
  N_deaths_eff <- r$c_yll   / c_mort

  c_health_daly <- N_true_eff   * c_morb_daly_blended
  c_yll_daly    <- N_deaths_eff * c_mort_daly
  c_society_daly <- r$c_regulator_tot + c_health_daly + c_yll_daly
  NMB_daly       <- c_society_SQ_daly - c_society_daly

  dalys_averted  <- r$deaths_averted * yll_per_death +
                    r$detected_true   * (p_sev * yld_severe + (1 - p_sev) * yld_mild) * 0.5
  # ^ 0.5: detected cases receive treatment — YLD reduced by ~half vs untreated

  r |> mutate(
    c_health_daly  = c_health_daly,
    c_yll_daly     = c_yll_daly,
    NMB_daly       = NMB_daly,
    dalys_averted  = dalys_averted,
    cost_per_daly  = if_else(dalys_averted > 0, c_regulator_tot / dalys_averted, NA_real_)
  )
}
